# samsungcloudplatformv2_organization_invitation (Resource)

Manages invitations for accounts to join the organization

## Example Usage

```terraform
provider "samsungcloudplatformv2" {
}

resource "samsungcloudplatformv2_organization_invitation" "invitation" {
  organization_id  = var.organization_id
  target_login_ids = var.target_login_ids
}

output "success_ids" {
  description = "Success IDs"
  value       = samsungcloudplatformv2_organization_invitation.invitation.success_ids
}

output "failed_ids" {
  description = "Failed IDs"
  value       = samsungcloudplatformv2_organization_invitation.invitation.failed_ids
}

variable "organization_id" {
  description = "Organization ID"
  type        = string
  default     = "ENTER YOUR RESOURCE'S ORGANIZATION_ID"
}

variable "target_login_ids" {
  description = "Target Login IDs to invite"
  type        = list(string)
  default     = ["ENTER YOUR RESOURCE'S TARGET_LOGIN_IDS"]
}
```
