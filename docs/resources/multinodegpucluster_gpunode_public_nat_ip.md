# samsungcloudplatformv2_multinodegpucluster_gpunode_public_nat_ip (Resource)

GPU Node Public NAT IP Resource. Assign(create)/Release(delete) a public NAT IP to a GPU Node.

## Example Usage

```terraform
provider "samsungcloudplatformv2" {
  default_region = "kr-west1"
}

resource "samsungcloudplatformv2_multinodegpucluster_gpunode_public_nat_ip" "public_nat_ip" {
  gpu_node_id          = var.gpu_node_id
  public_ip_address_id = var.public_ip_address_id
  timeouts {
    create = var.create_timeouts
    delete = var.delete_timeouts
  }
}



output "gpunode_public_nat_ip_output" {
  value = samsungcloudplatformv2_multinodegpucluster_gpunode_public_nat_ip.public_nat_ip
}


variable "gpu_node_id" {
  type    = string
  default = "ENTER YOUR RESOURCE'S GPU_NODE_ID"
}

variable "public_ip_address_id" {
  type    = string
  default = "ENTER YOUR RESOURCE'S PUBLIC_IP_ADDRESS_ID"
}

variable "create_timeouts" {
  type    = string
  default = "20m"
}

variable "delete_timeouts" {
  type    = string
  default = "20m"
}
```
