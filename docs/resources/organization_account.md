# samsungcloudplatformv2_organization_account (Resource)

Manages accounts within an organization

## Example Usage

```terraform

}

  login_id        = var.login_id
  name            = var.name
  organization_id = var.organization_id
  role_name       = var.role_name
  lazy_policy     = var.lazy_policy
  parent_unit_id  = var.parent_unit_id
}

output "success" {
  description = "Create Success Info"
  value       = samsungcloudplatformv2_organization_account.account.success
}

output "failed" {
  description = "Create Failed Info"
  value       = samsungcloudplatformv2_organization_account.account.failed

  description = "Login ID"

  default     = "ENTER YOUR RESOURCE'S LOGIN_ID"
```
